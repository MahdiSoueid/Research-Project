typedef unsigned cell_t; // // “small cell” typedef unsigned char scell_t;

static cell_t *_table = NULL, *_alternate_table = NULL;

// lazy change tables (indexed by tile) // number of tiles static size_t _ntx; static size_t _nty; static scell_t *_active_table = NULL; static scell_t *_change_table = NULL;

static inline cell_t *table_cell (cell_t *restrict i, int y, int x) { return i + y * DIM + x; }

static inline scell_t *table_scell (scell_t *restrict i, int ty, int tx) { return i + ty * _ntx + tx; } // This kernel does not directly work on cur_img/next_img. // Instead, we use 2D arrays of boolean values, not colors /#define cur_table(y, x) (*table_cell (_table, (y), (x))) /#define next_table(y, x) (*table_cell (_alternate_table, (y), (x)))

// similar idea for lazy change-checking /#define active_tile(ty, tx) (*table_scell (_active_table, ty, tx)) /#define changed_tile(ty, tx) (*table_scell (_change_table, ty, tx))

static inline void activate_neighbors (int ty, int tx) { for (int dy = -1; dy ⇐ 1; dy++) for (int dx = -1; dx ⇐ 1; dx++) { int ny = ty + dy; int nx = tx + dx; if (nx >= 0 && nx < _ntx && ny >= 0 && ny < _nty) active_tile (ny, nx) = true; } }

void lazy_init (void) { _ntx = DIM / TILE_W; _nty = DIM / TILE_H; const size_t size = _ntx * _nty * sizeof (scell_t);

_active_table = mmap (NULL, size, PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);

_change_table = mmap (NULL, size, PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);

for (size_t t = 0; t < _ntx * _nty; ++t) { _active_table [t] = 1; } }

void life_init (void) { // life_init may be (indirectly) called several times so we check if data were // already allocated if (_table == NULL) { const unsigned size = DIM * DIM * sizeof (cell_t);

_table = mmap (NULL, size, PROT_READ | PROT_WRITE,
               MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);

_alternate_table = mmap (NULL, size, PROT_READ | PROT_WRITE,
                         MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);

} }

void life_finalize (void) { const unsigned size = DIM * DIM * sizeof (cell_t);

munmap (_table, size); munmap (_alternate_table, size);

if (_active_table != NULL) { munmap (_active_table, _ntx * _nty * sizeof (scell_t)); } if (_change_table != NULL) { munmap (_change_table, _ntx * _nty * sizeof (scell_t)); } }

// This function is called whenever the graphical window needs to be refreshed void life_refresh_img (void) { for (int i = 0; i < DIM; i++) for (int j = 0; j < DIM; j++) cur_img (i, j) = cur_table (i, j) * LIFE_COLOR; }

static inline void swap_tables (void) { cell_t *tmp = _table;

_table = _alternate_table; _alternate_table = tmp; }

///////////////////////////// Default tiling int life_do_tile_default (int x, int y, int width, int height) { int change = 0;

for (int i = y; i < y + height; i++) for (int j = x; j < x + width; j++) if (j > 0 && j < DIM - 1 && i > 0 && i < DIM - 1) {

    unsigned n  = 0;
    unsigned me = cur_table (i, j);

    for (int yloc = i - 1; yloc < i + 2; yloc++)
      for (int xloc = j - 1; xloc < j + 2; xloc++)
        if (xloc != j || yloc != i)
          n += cur_table (yloc, xloc);

    if (me == 1 && n != 2 && n != 3) {
      me     = 0;
      change = 1;
    } else if (me == 0 && n == 3) {
      me     = 1;
      change = 1;
    }

    next_table (i, j) = me;
  }

return change; }

///////////////////////////// Sequential version (seq) // unsigned life_compute_seq (unsigned nb_iter) { for (unsigned it = 1; it ⇐ nb_iter; it++) {

int change = do_tile (0, 0, DIM, DIM);

if (!change)
  return it;

swap_tables ();

}

return 0; }

///////////////////////////// Tiled sequential version (tiled) // unsigned life_compute_tiled (unsigned nb_iter) { unsigned res = 0;

for (unsigned it = 1; it ⇐ nb_iter; it++) { unsigned change = 0;

for (int y = 0; y < DIM; y += TILE_H)
  for (int x = 0; x < DIM; x += TILE_W)
    change |= do_tile (x, y, TILE_W, TILE_H);

swap_tables ();

if (!change) { // we stop if all cells are stable
  res = it;
  break;
}

}

return res; }

/** *

  • openMP version of the tiled implementation

*/ unsigned life_compute_openmp (unsigned nb_iter) { unsigned res = 0;

for (unsigned it = 1; it ⇐ nb_iter; it++) { unsigned change = 0;

/#pragma omp parallel for collapse(2) reduction(| : change) for (int y = 0; y < DIM; y += TILE_H) for (int x = 0; x < DIM; x += TILE_W) change |= do_tile (x, y, TILE_W, TILE_H);

swap_tables ();

if (!change) { // we stop if all cells are stable
  res = it;
  break;
}

}

return res; }

/** *

  • lazy openMP version of the tiled implementation

*/ unsigned life_compute_lazy (unsigned nb_iter) { if (_active_table == NULL) { lazy_init (); } unsigned res = 0;

for (unsigned it = 1; it ⇐ nb_iter; it++) { unsigned change = 0;

/#pragma omp parallel { /#pragma omp for collapse(2) reduction(| : change)
schedule(guided, 8 * TILE_H * TILE_W) for (int ty = 0; ty < _nty; ++ty) for (int tx = 0; tx < _ntx; ++tx) { // reset changed changed_tile (ty, tx) = 0; // only compute do_tile on possibly changed tiles if (active_tile (ty, tx)) { changed_tile (ty, tx) = do_tile (tx * TILE_W, ty * TILE_H, TILE_W, TILE_H);

        change |= changed_tile (ty, tx);
        // deactivate it before computing next timestep active_tile state
        active_tile (ty, tx) = 0;
      }
    }
}
swap_tables ();

if (!change) { // we stop if all cells are stable
  res = it;
  break;
}
// update active_tile sequentially ( 9 * (number of changed tiles) writes)
for (int ty = 0; ty < _nty; ++ty) {
  for (int tx = 0; tx < _ntx; ++tx) {
    if (changed_tile (ty, tx)) {
      activate_neighbors (ty, tx);
    }
  }
}

}

return res; }
